# Q7-14: Async examples

- 1

```csharp
public async Task Main()
{
    Console.WriteLine("Sync 1-1"); //1 - Thread 1

    await Async1(); //2 - Thread 1

    Console.WriteLine("Sync 1-2"); //6 - Thread 2
}

public async Task Async1()
{
    Console.WriteLine("Sync 2-1"); //3 - Thread 1

    await Task.Delay(1000); //4 - Thread 1 --> Thread 1 releases for other requests.

    Console.WriteLine("Sync 2-2"); //5 - Thread 2
}
```

- 2

```csharp
public async Task Main()
{
    Console.WriteLine("Sync 1-1"); //1 - Thread 1

    var task1 = Async1(); //2 - Thread 1
    var task2 = Async2(); //5 - Thread 1

    Console.WriteLine("Sync 1-2"); //8 - Thread 1

    await task1; //9 - Thread 1 --> Thread 1 releases for other requests.
    await task2; //11 - Thread 2 --> Thread 2 releases for other requests.

    Console.WriteLine("Sync 1-2"); //13 - Thread 3
}

public async Task Async1()
{
    Console.WriteLine("Sync 2-1"); //3 - Thread 1

    await Task.Delay(1000); //4 - Thread 1

    Console.WriteLine("Sync 2-2"); //10 - Thread 2
}

public async Task Async2()
{
    Console.WriteLine("Sync 3-1"); //6 - Thread 1

    await Task.Delay(1000); //7 - Thread 1

    Console.WriteLine("Sync 3-2"); //12 - Thread 3
}
```

- 3

```csharp
public async Task Main()
{
    Console.WriteLine("Sync 1-1"); //1 - Thread 1

    var task1 = Async1(); //2 - Thread 1
    var task2 = Async2(); //5 - Thread 1

    Console.WriteLine("Sync 1-2"); //8 - Thread 1

    await Task.WhenAll(task1, task2); //9 - Thread 1 --> Thread 1 releases for other requests.

    Console.WriteLine("Sync 1-2"); //12 - Thread 2
}

public async Task Async1()
{
    Console.WriteLine("Sync 2-1"); //3 - Thread 1

    await Task.Delay(200); //4 - Thread 1

    Console.WriteLine("Sync 2-2"); //11 - Thread 2
}

public async Task Async2()
{
    Console.WriteLine("Sync 3-1"); //6 - Thread 1

    await Task.Delay(100); //7 - Thread 1

    Console.WriteLine("Sync 3-2"); //10 - Thread 2
}
```
