# Q3-9: What is the difference between Autofac and built-in .Net DI container?

- You can use Autofac instead of or alongside the built-in container in .NET.
- It works similarly to the built-in ASP.NET Core DI, but offers more advanced features, like:

  - Conditional registrations
  - Property injection
  - Module-based configuration
  - Better control over lifetime and scopes

- Example:

  1. Using Built-in DI Container

```csharp
public interface INotificationService
{
    void Notify(string message);
}

public class EmailNotificationService : INotificationService
{
    public void Notify(string message)
    {
        Console.WriteLine($"Email sent: {message}");
    }
}

builder.Services.AddScoped<INotificationService, EmailNotificationService>();

public class HomeController : Controller
{
    private readonly INotificationService _notifier;

    public HomeController(INotificationService notifier)
    {
        _notifier = notifier;
    }

    public IActionResult Index()
    {
        _notifier.Notify("Welcome!");
        return View();
    }
}
```

  2. Using Autofac

```csharp
builder.Host.UseServiceProviderFactory(new AutofacServiceProviderFactory());
builder.Host.ConfigureContainer<ContainerBuilder>(container =>
{
    container.RegisterType<EmailNotificationService>()
             .As<INotificationService>()
             .InstancePerLifetimeScope();
});
```

- Conditional injection:

    1. Autofac: It gives you built-in, cleaner support

```csharp
public interface INotificationService
{
    void Notify(string message);
}

public class EmailNotificationService : INotificationService
{
    public void Notify(string message)
    {
        Console.WriteLine($"Email: {message}");
    }
}

public class SmsNotificationService : INotificationService
{
    public void Notify(string message)
    {
        Console.WriteLine($"SMS: {message}");
    }
}

public interface INotificationFactory
{
    INotificationService Create(bool isVip);
}

public class NotificationFactory : INotificationFactory
{
    private readonly IComponentContext _context;

    public NotificationFactory(IComponentContext context)
    {
        _context = context;
    }

    public INotificationService Create(bool isVip)
    {
        if (isVip)
            return _context.ResolveNamed<INotificationService>("sms");

        return _context.ResolveNamed<INotificationService>("email");
    }
}

builder.Host.UseServiceProviderFactory(new AutofacServiceProviderFactory());
builder.Host.ConfigureContainer<ContainerBuilder>(container =>
{
    container.RegisterType<EmailNotificationService>()
             .Named<INotificationService>("email");

    container.RegisterType<SmsNotificationService>()
             .Named<INotificationService>("sms");

    container.RegisterType<NotificationFactory>()
             .As<INotificationFactory>()
             .SingleInstance();
});

public class HomeController : Controller
{
    private readonly INotificationFactory _factory;

    public HomeController(INotificationFactory factory)
    {
        _factory = factory;
    }

    public IActionResult Index()
    {
        var isVip = true;
        var notifier = _factory.Create(isVip);
        notifier.Notify("Welcome to the site!");

        return View();
    }
}
```

  2. Built-in Container: 

```csharp

public interface INotificationService
{
    void Notify(string message);
}

public class EmailNotificationService : INotificationService
{
    public void Notify(string message)
    {
        Console.WriteLine($"Email: {message}");
    }
}

public class SmsNotificationService : INotificationService
{
    public void Notify(string message)
    {
        Console.WriteLine($"SMS: {message}");
    }
}

public interface INotificationFactory
{
    INotificationService Create(bool isVip);
}

public class NotificationFactory : INotificationFactory
{
    private readonly EmailNotificationService _email;
    private readonly SmsNotificationService _sms;

    public NotificationFactory(EmailNotificationService email, SmsNotificationService sms)
    {
        _email = email;
        _sms = sms;
    }

    public INotificationService Create(bool isVip)
    {
        return isVip ? _sms : _email;
    }
}

builder.Services.AddSingleton<EmailNotificationService>();
builder.Services.AddSingleton<SmsNotificationService>();
builder.Services.AddSingleton<INotificationFactory, NotificationFactory>();

public class HomeController : Controller
{
    private readonly INotificationFactory _factory;

    public HomeController(INotificationFactory factory)
    {
        _factory = factory;
    }

    public IActionResult Index()
    {
        var isVip = true;
        var notifier = _factory.Create(isVip);
        notifier.Notify("Welcome!");
        return View();
    }
}
```
