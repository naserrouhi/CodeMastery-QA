- It is a routing mechanism that directs messages to one or more queues based on routing rules.

- Types of Exchanges in RabbitMQ:

    1. Direct Exchange (Point-to-Point)

        - Routes messages to queues with an exact matching routing key.
        - Example: A payment service sending different types of notifications, where each notification type has its own queue, like payment.success or payment.failed.

```csharp
class DirectExchangeProducer
{
    static void Main()
    {
        var factory = new ConnectionFactory() { HostName = "localhost" };
        using var connection = factory.CreateConnection();
        using var channel = connection.CreateModel();

        var exchangeName = "direct_logs";
        channel.ExchangeDeclare(exchange: exchangeName, type: ExchangeType.Direct);

        var routingKey = "payment.success";
        var message = "Payment processed successfully";

        var body = Encoding.UTF8.GetBytes(message);
        channel.BasicPublish(exchange: exchangeName, routingKey: routingKey, basicProperties: null, body: body);
        
        Console.WriteLine($"Sent: {message}");
    }
}

class DirectExchangeConsumer
{
    static void Main()
    {
        var factory = new ConnectionFactory() { HostName = "localhost" };
        using var connection = factory.CreateConnection();
        using var channel = connection.CreateModel();

        var exchangeName = "direct_logs";
        channel.ExchangeDeclare(exchange: exchangeName, type: ExchangeType.Direct);

        var queueName = channel.QueueDeclare().QueueName;
        var routingKey = "payment.success";

        channel.QueueBind(queue: queueName, exchange: exchangeName, routingKey: routingKey);

        var consumer = new EventingBasicConsumer(channel);
        consumer.Received += (model, ea) =>
        {
            var message = Encoding.UTF8.GetString(ea.Body.ToArray());
            Console.WriteLine($"Received: {message}");
        };

        channel.BasicConsume(queue: queueName, autoAck: true, consumer: consumer);

        Console.WriteLine("Waiting for messages...");
        Console.ReadLine();
    }
}
```

    2. Fanout Exchange (Broadcast)

        - Routes messages to all queues bound to it, ignoring the routing key.
        - Example: A news service broadcasting breaking news to many queues where each consumer may display news on different platforms.

```csharp
class FanoutExchangeProducer
{
    static void Main()
    {
        var factory = new ConnectionFactory() { HostName = "localhost" };
        using var connection = factory.CreateConnection();
        using var channel = connection.CreateModel();

        var exchangeName = "logs";
        channel.ExchangeDeclare(exchange: exchangeName, type: ExchangeType.Fanout);

        var message = "System is up and running";
        var body = Encoding.UTF8.GetBytes(message);

        channel.BasicPublish(exchange: exchangeName, routingKey: "", basicProperties: null, body: body);
        Console.WriteLine($"Sent: {message}");
    }
}

class FanoutExchangeConsumer
{
    static void Main()
    {
        var factory = new ConnectionFactory() { HostName = "localhost" };
        using var connection = factory.CreateConnection();
        using var channel = connection.CreateModel();

        var exchangeName = "logs";
        channel.ExchangeDeclare(exchange: exchangeName, type: ExchangeType.Fanout);

        var queueName = channel.QueueDeclare().QueueName;
        channel.QueueBind(queue: queueName, exchange: exchangeName, routingKey: "");

        var consumer = new EventingBasicConsumer(channel);
        consumer.Received += (model, ea) =>
        {
            var message = Encoding.UTF8.GetString(ea.Body.ToArray());
            Console.WriteLine($"Received: {message}");
        };

        channel.BasicConsume(queue: queueName, autoAck: true, consumer: consumer);

        Console.WriteLine("Waiting for messages...");
        Console.ReadLine();
    }
}
```        

    3. Topic Exchange (Pattern-based)

        - Routes messages to queues based on pattern matching in the routing key.
        - Example: A log monitoring system where logs from different services are sent to different queues, like logs.*.database or logs.#.critical.

```csharp
class TopicExchangeProducer
{
    static void Main()
    {
        var factory = new ConnectionFactory() { HostName = "localhost" };
        using var connection = factory.CreateConnection();
        using var channel = connection.CreateModel();

        var exchangeName = "topic_logs";
        channel.ExchangeDeclare(exchange: exchangeName, type: ExchangeType.Topic);

        var routingKey = "logs.error.database";
        var message = "Database connection error";

        var body = Encoding.UTF8.GetBytes(message);
        channel.BasicPublish(exchange: exchangeName, routingKey: routingKey, basicProperties: null, body: body);

        Console.WriteLine($"Sent: {message}");
    }
}

class TopicExchangeConsumer
{
    static void Main()
    {
        var factory = new ConnectionFactory() { HostName = "localhost" };
        using var connection = factory.CreateConnection();
        using var channel = connection.CreateModel();

        var exchangeName = "topic_logs";
        channel.ExchangeDeclare(exchange: exchangeName, type: ExchangeType.Topic);

        var queueName = channel.QueueDeclare().QueueName;

        var routingKeyPattern = "logs.#";
        channel.QueueBind(queue: queueName, exchange: exchangeName, routingKey: routingKeyPattern);

        var consumer = new EventingBasicConsumer(channel);
        consumer.Received += (model, ea) =>
        {
            var message = Encoding.UTF8.GetString(ea.Body.ToArray());
            var routingKey = ea.RoutingKey;
            Console.WriteLine($"Received: '{message}' with routing key: '{routingKey}'");
        };

        channel.BasicConsume(queue: queueName, autoAck: true, consumer: consumer);

        Console.WriteLine("Waiting for messages...");
        Console.ReadLine();
    }
}
```        

    4. Headers Exchange (Attribute-based)

        - Routes messages based on message header attributes instead of the routing key.
        - Example: An order management system where different queues receive orders based on the region or priority header.

```csharp
class HeadersExchangeProducer
{
    static void Main()
    {
        var factory = new ConnectionFactory() { HostName = "localhost" };
        using var connection = factory.CreateConnection();
        using var channel = connection.CreateModel();

        var exchangeName = "headers_logs";
        channel.ExchangeDeclare(exchange: exchangeName, type: ExchangeType.Headers);

        var message = "Urgent system alert";
        var body = Encoding.UTF8.GetBytes(message);

        var headers = new Dictionary<string, object>
        {
            { "priority", "high" },
            { "region", "US" }
        };

        var properties = channel.CreateBasicProperties();
        properties.Headers = headers;

        channel.BasicPublish(exchange: exchangeName, routingKey: "", basicProperties: properties, body: body);

        Console.WriteLine($"Sent: {message}");
    }
}

class HeadersExchangeConsumer
{
    static void Main()
    {
        var factory = new ConnectionFactory() { HostName = "localhost" };
        using var connection = factory.CreateConnection();
        using var channel = connection.CreateModel();

        var exchangeName = "headers_logs";
        channel.ExchangeDeclare(exchange: exchangeName, type: ExchangeType.Headers);

        var queueName = channel.QueueDeclare().QueueName;

        var headerArguments = new Dictionary<string, object>
        {
            { "priority", "high" }
        };
        
        channel.QueueBind(queue: queueName, exchange: exchangeName, routingKey: "", arguments: headerArguments);

        var consumer = new EventingBasicConsumer(channel);
        consumer.Received += (model, ea) =>
        {
            var message = Encoding.UTF8.GetString(ea.Body.ToArray());
            var headers = ea.BasicProperties.Headers;
            var priority = headers.ContainsKey("priority") ? headers["priority"].ToString() : "N/A";
            Console.WriteLine($"Received: '{message}' with priority: '{priority}'");
        };

        channel.BasicConsume(queue: queueName, autoAck: true, consumer: consumer);

        Console.WriteLine("Waiting for messages...");
        Console.ReadLine();
    }
}
```
