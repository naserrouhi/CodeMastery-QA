# Q2-8: How can we change interface registration dynamically?

```csharp
public interface ISmsService
{
    void SendSms(string phoneNumber, string message);
}

public class SmsProviderA : ISmsService
{
    public void SendSms(string phoneNumber, string message)
    {
        Console.WriteLine($"Sending SMS via Provider A: {message} to {phoneNumber}");
    }
}

public class SmsProviderB : ISmsService
{
    public void SendSms(string phoneNumber, string message)
    {
        Console.WriteLine($"Sending SMS via Provider B: {message} to {phoneNumber}");
    }
}
```

1. Dynamic Registration (Simple Approach)

    - This approach is best for small-scale applications or simple use cases. It tightly couples the resolution logic to the DI container registration.

```csharp
builder.Services.AddSingleton<SmsProviderA>();
builder.Services.AddSingleton<SmsProviderB>();

builder.Services.AddScoped<ISmsService>(provider =>
{
    var configuration = provider.GetRequiredService<IConfiguration>();
    var providerName = configuration["SmsProvider"];

    return providerName switch
    {
        "ProviderA" => provider.GetRequiredService<SmsProviderA>(),
        "ProviderB" => provider.GetRequiredService<SmsProviderB>(),
        _ => throw new InvalidOperationException("Invalid SMS Provider configuration")
    };
});
```

```csharp
public class SmsServiceConsumer
{
    private readonly ISmsService _smsService;

    public SmsServiceConsumer(ISmsService smsService)
    {
        _smsService = smsService;
    }

    public void SendSms(string phoneNumber, string message)
    {
        _smsService.SendSms(phoneNumber, message);
    }
}
```

2. Factory Pattern (Flexible Approach)

    - This approach is better for larger applications or when you need more control over object creation. It adheres to the Single Responsibility Principle.

```csharp
public class SmsServiceFactory
{
    private readonly IServiceProvider _serviceProvider;
    private readonly IConfiguration _configuration;

    public SmsServiceFactory(IServiceProvider serviceProvider, IConfiguration configuration)
    {
        _serviceProvider = serviceProvider;
        _configuration = configuration;
    }

    public ISmsService GetSmsService()
    {
        var providerName = _configuration["SmsProvider"];
        return providerName switch
        {
            "ProviderA" => _serviceProvider.GetRequiredService<SmsProviderA>(),
            "ProviderB" => _serviceProvider.GetRequiredService<SmsProviderB>(),
            _ => throw new InvalidOperationException("No valid provider found")
        };
    }
}
```

```csharp
builder.Services.AddSingleton<SmsProviderA>();
builder.Services.AddSingleton<SmsProviderB>();
builder.Services.AddSingleton<SmsServiceFactory>();
```

```csharp
public class SmsServiceConsumer
{
    private readonly SmsServiceFactory _smsServiceFactory;

    public SmsServiceConsumer(SmsServiceFactory smsServiceFactory)
    {
        _smsServiceFactory = smsServiceFactory;
    }

    public void SendSms(string phoneNumber, string message)
    {
        var smsService = _smsServiceFactory.GetSmsService();
        smsService.SendSms(phoneNumber, message);
    }
}
```
